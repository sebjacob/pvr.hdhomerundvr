# ============================================================================
# Changelog / addon news
# ============================================================================

# Keep this deliberately simple and XML-safe.
#
# The addon.xml template receives this through @PVR_CHANGELOG@.
#
# Do NOT put Markdown links or XML markup into this variable.

set(PVR_CHANGELOG
  "v${PVR_VERSION_STRING}\n  - Android build"
)

# If a real changelog exists, use it instead.
foreach(CHANGELOG_CANDIDATE
  CHANGELOG.md
  CHANGELOG.txt
  changelog.txt
  NEWS.md
  NEWS.txt
)

  if(EXISTS "${PROJECT_ROOT}/${CHANGELOG_CANDIDATE}")

    file(READ
      "${PROJECT_ROOT}/${CHANGELOG_CANDIDATE}"
      PVR_CHANGELOG
    )

    break()

  endif()

endforeach()

# Make changelog safe for use as XML element text.
string(REPLACE "&" "&amp;" PVR_CHANGELOG "${PVR_CHANGELOG}")
string(REPLACE "<" "&lt;"  PVR_CHANGELOG "${PVR_CHANGELOG}")
string(REPLACE ">" "&gt;"  PVR_CHANGELOG "${PVR_CHANGELOG}")

# ============================================================================
# Assemble Kodi addon directory
# ============================================================================

add_custom_command(
  TARGET hdhomerundvr
  POST_BUILD

  COMMAND
    "${CMAKE_COMMAND}" -E make_directory
    "${PVR_PACKAGE_DIR}"

  COMMAND
    "${CMAKE_COMMAND}" -E make_directory
    "${PVR_PACKAGE_RESOURCES_DIR}"

  COMMAND
    "${CMAKE_COMMAND}" -E make_directory
    "${PVR_PACKAGE_RESOURCES_DIR}/language/resource.language.en_gb"

  COMMAND
    "${CMAKE_COMMAND}" -E copy_if_different
    "${PVR_LICENSE}"
    "${PVR_PACKAGE_DIR}/LICENSE"

  COMMAND
    "${CMAKE_COMMAND}" -E copy_if_different
    "${PVR_NOMEDIA}"
    "${PVR_PACKAGE_DIR}/.nomedia"

  COMMAND
    "${CMAKE_COMMAND}" -E copy_if_different
    "${PVR_ICON}"
    "${PVR_PACKAGE_RESOURCES_DIR}/icon.png"

  COMMAND
    "${CMAKE_COMMAND}" -E copy_if_different
    "${PVR_SETTINGS}"
    "${PVR_PACKAGE_RESOURCES_DIR}/settings.xml"

  COMMAND
    "${CMAKE_COMMAND}" -E copy_if_different
    "${PVR_STRINGS}"
    "${PVR_PACKAGE_RESOURCES_DIR}/language/resource.language.en_gb/strings.po"

  COMMAND
    "${CMAKE_COMMAND}" -E copy
    "$<TARGET_FILE:hdhomerundvr>"
    "${PVR_PACKAGE_DIR}/libhdhomerundvr.so"

  COMMAND
    "${CMAKE_STRIP}"
    --strip-unneeded
    "${PVR_PACKAGE_DIR}/libhdhomerundvr.so"

  COMMENT
    "Assembling Kodi Android addon"

  VERBATIM
)

# ============================================================================
# Verify generated addon.xml and package
# ============================================================================

add_custom_command(
  TARGET hdhomerundvr
  POST_BUILD

  COMMAND
    "${CMAKE_COMMAND}" -E echo
    "Kodi addon package assembled: ${PVR_PACKAGE_DIR}"

  COMMAND
    "${CMAKE_COMMAND}" -E echo
    "Checking addon.xml"

  COMMAND
    "${CMAKE_COMMAND}" -E cat
    "${PVR_PACKAGE_DIR}/addon.xml"

  COMMENT
    "Verifying Kodi addon package"

  VERBATIM
)

# ============================================================================
# Final ZIP
# ============================================================================

add_custom_target(package
  ALL

  DEPENDS
    hdhomerundvr

  COMMAND
    "${CMAKE_COMMAND}" -E rm -f
    "${PVR_ZIP}"

  COMMAND
    "${CMAKE_COMMAND}" -E tar
    "cf"
    "${PVR_ZIP}"
    --format=zip
    "pvr.hdhomerundvr"

  WORKING_DIRECTORY
    "${PVR_OUTPUT_DIR}"

  COMMENT
    "Creating Kodi addon ZIP"

  VERBATIM
)
